PROCEDURE txt_update

select invoice
if empty(invoice.amount_sub) or empty(invoice.amount_tot) or empty(invoice.amount_bal) or (invoice.amount_pay # minvpay)
   *minvsub  = VinvSub('VAL')
   *minvbal  = Vinvbal('VAL')
   store 0 to msub, mpph, mvat, mtot
   select lineitem
   set order to invoice
   seek invoice.invoice
   scan while lineitem.invoice = invoice.invoice

     invoice.amount_pph with mpph,;
     invoice.amount_vat with mvat,;
     invoice.amount_tot with mtot,;
     invoice.amount_pay with minvpay,;

PROCEDURE check_editable
do Case
   case eof() or bof()
      =MESSAGEBOX("Invoice file is empty.",0+16, "Command Warning")
      editable = .f.
   case not empty(invoice.voided)
      =MESSAGEBOX("Invoice has been voided.",0+16, "Command Warning")
      editable = .f.
   case not empty(invoice.posted)
      =MESSAGEBOX("Invoice has been posted.",0+16, "Command Warning")
      editable = .f.
   case not empty(invoice.printed)  and _pass[7]
      =MESSAGEBOX("Invoice has been printed.",0+16, "Command Warning")
      editable = .f.
   *case alltrim(invoice.operator) # alltrim(_userid)  and _pass[7]
   *   =MESSAGEBOX("You are not " + alltrim(invoice.operator) + " !",0+16, "Command Warning")
   *   editable = .f.
   case Vinvpay() # 0     && and _pass[7]
      =MESSAGEBOX("Invoice has payment.",0+16, "Command Warning")
      editable = .f.
   otherwise
      editable = .t.
endcase
ENDPROC

PROCEDURE so_match
wait 'Matching data to SO ...' window nowait
if invoice.divisi='*'
   return
endif
* First, clear any existing invoice number in soline
select 0
use lineitem alias ar_line
set order to lns_itm
wait 'Deleting existing invoice no ' + invoice.invoice window nowait
select sodtl
set order to vinvoice
seek invoice.invoice
do while found()
   replace vinvoice with '', cost with 0, rate with 1, vendcode with ''
   seek invoice.invoice
enddo
set order to order
select lineitem
seek invoice.invoice
scan while invoice = invoice.invoice
   select sodtl
   seek lineitem.order + lineitem.itemcode
   if found()
      replace ;
        vinvoice with lineitem.invoice,;
        cost     with lineitem.cost,;
        rate     with lineitem.rate,;
        vendcode with lineitem.vendcode
      if not empty(sodtl.invoice)
         select ar_line
         seek sodtl.invoice + sodtl.itemcode
         if found()
            replace;
               ar_line.cost  with sodtl.cost,;
               ar_line.rate  with sodtl.rate,;
               ar_line.vinvoice with sodtl.vinvoice
         endif
      endif
      select sodtl
   else
      wait 'SO No.'+lineitem.order + ' - ' + lineitem.itemcode + " doesn't exists." window timeout 3
   endif
   select lineitem
endscan
select ar_line
select lineitem
ENDPROC

endif
maction = iif(empty(invoice.posted), 'POST', 'UN-POST')
mauthorized = .f.
no_variance = .t.
mright = 9
if not empty(invoice.posted)
   no_variance = .f.
   mright = 12
endif
if no_variance
   do form Authorize with 'vinvoice_log.dbf','invoice', maction, invoice.invoice, "INVOICE" to mauthorized
else
   do form Authorize_ex with 'vinvoice_log.dbf','invoice', maction, invoice.invoice, "INVOICE", maction + " AP", mright  to mauthorized 
endif   
*nAnswer=MESSAGEBOX("Do you want to " + maction + ", now ?",4+32, "Command Warning")
*if nAnswer # 6
*   =MESSAGEBOX("Can't " + maction + " : you have cancelled the procedure.",0+16, "Command Warning")
*   return
*endif
if mauthorized
   thisform.status_off
   wait "Please wait, updating ledger..." window nowait
   select invoice
   minvoice = invoice.invoice
   if maction = 'POST'
      if GetPeriod(invoice.written) # invoice.period
         =MESSAGEBOX("Error: Wrong Period !!! Please check your PERIOD and DATE first.",0+48, "Command Warning")
         return
      endif
      = Post2Gl ('POST AP INVOICE')
      *= fLogs_add('vinvoice_log.dbf', invoice.invoice, 'POST', '')
   else
      = Post2Gl('REVERSE AP INVOICE')
      *= fLogs_add('vinvoice_log.dbf', invoice.invoice, 'POST', '')
   endif
   select invoice
   seek minvoice
   if maction = 'POST'
      replace invoice.posted with date()
   else
      replace invoice.posted with ctod("  /  /    ")
   endif
   =MESSAGEBOX("AP Invoice has been " + maction + "ED.",0+48, "Command Warning")
   thisform.status_on
endif   
thisform.txt_update
thisform.refresh
ENDPROC
