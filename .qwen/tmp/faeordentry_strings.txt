PROCEDURE append_item
select ae_ordline
set order to order
seek ae_order.order
if not found()
   select ae_item
   set order to itemcode
   go top
   scan
      if ae_item.divisi = ae_pitem
         SELECT ae_ordline
         APPEND BLANK
         replace ;
               ae_ordline.itemcode with ae_item.itemcode,;
               ae_ordline.descript with ae_item.descript,;
               ae_ordline.order    with ae_noorder,;
               ae_ordline.glrev    with ae_item.glrev,;
               ae_ordline.glcogs   with ae_item.glcogs,;
               ae_ordline.amount   with ae_item.price,;
               ae_ordline.valas    with ae_item.valas,;
               ae_ordline.taxable  with ae_item.taxable,;
               ae_ordline.cost     with 0,;
               ae_ordline.rate     with iif(ae_item.valas = _valas, 1, FRate(ae_item.valas)),;
               ae_ordline.arate    with iif(ae_item.valas = _valas, 1, FRate(ae_item.valas)),;
               ae_ordline.posted   with .f.
         SELECT ae_item
      ENDIF 
   endscan
endif

PROCEDURE update_detail
*******************update cost**********
select 0
use trdtl again alias ftrdtl
set order to order
select 0
use dndtl
set order to order
if used('lineitem')
   select lineitem
   linopen = .t.
else
   SELECT 0
   USE lineitem
   linopen = .f.
endif
if used('vline')
   select vline
   vinopen = .t.
else
   SELECT 0
   USE vline
   vinopen = .f.
endif
fpvfile = ae_pflag + 'pv.dbf'
wait fpvfile window timeout 3
select 0
use (fpvfile) alias fpv
set order to order
select ae_order
morder_  = order
mdivisi_ = divisi
select ae_ordline
seek morder_
scan while order = morder_
   wait "Please wait, update cost..." window nowait
   select ae_ordline
   mitem = itemcode
   replace cost with 0, amount with 0, pv with '', invoice with '', ntr with '', vinvoice with '', posted with .f.
   select ftrdtl
   set order to order
   seek mdivisi_ + morder_ + mitem
   if found()
      xtr   = iif(ftrdtl.lcancel, '', ftrdtl.ntr)
      xcost = iif(ftrdtl.lcancel, 0, ftrdtl.tr_cost)
      
      select ae_ordline
      replace ntr with iif(ftrdtl.lposted, '', xtr), cost with iif(ftrdtl.lposted, 0, xcost)
   endif
   select fpv
   set order to order
   seek morder_ + mitem
   if found() and not empty(posted)
      mcost = fpv.cost
      xpv   = fpv.number
      xrate = fpv.rate
      xntr  = fpv.ntr
      
      select ae_ordline
      replace cost with mcost, pv with xpv, rate with xrate, ntr  with xntr
      if fpv.itemcode = 'SI020' or fpv.itemcode = 'SI034' or fpv.itemcode = 'SE020' or fpv.itemcode = 'SE034' or fpv.itemcode = 'SC020' or fpv.itemcode = 'SC034'  &&Deposit containter cost dihilangkan
         select ae_ordline
         replace amount with mcost
      endif
      if fpv.itemcode = 'SI520' or fpv.itemcode = 'SI534' or fpv.itemcode = 'SE520' or fpv.itemcode = 'SE534' or fpv.itemcode = 'SC520' or fpv.itemcode = 'SC534'  &&Deposit containter cost dihilangkan
         select ae_ordline
         replace amount with mcost
      endif
   endif
   select lineitem
   set order to LNS_ORD
   seek mdivisi_ + morder_ + mitem
   if found()
      mamount  = lineitem.amt_orig
      xinvoice = lineitem.invoice
      mrate
   = lineitem.rate_job
      
      select ae_ordline
      replace invoice with xinvoice, amount with mamount, arate with mrate, posted with .t.
   endif
   select vline
   set order to LNS_ORD
   seek mdivisi_ + morder_ + mitem
   if found()
      mcost = vline.cost
      xpv   = vline.invoice
      xrate = vline.rate
      
      select ae_ordline
      replace cost with mcost, vinvoice with xpv, rate with xrate 
   endif
   select dndtl
   set order to order
   seek mdivisi_ + morder_ + mitem
   if found()
      mamount = dndtl.debit
      mcost   = dndtl.credit
      xpv     = dndtl.batch
      xrate   = dndtl.rate
      
      select ae_ordline
      replace amount with mamount, cost with mcost, dn with xpv, arate with xrate, rate with xrate 
   endif
   select ae_ordline
endscan
select fpv
select ftrdtl
select dndtl
if not linopen
   select lineitem
   use
endif
if not vinopen
   select vline
   use
endif
ENDPROC
