PROCEDURE cmdOk.Click
* CHECK VALIDITY
mdate = thisform.txtWritten.value
select sohdr
locate for lposted
if not found()
   =MESSAGEBOX("You have cancelled posting order procedure.",0+48, "Command Warning")
else
   nAnswer=MESSAGEBOX("Post all 'TAG' order, now ?", 4+32, "Command Warning")
   if nAnswer # 6
      =MESSAGEBOX("You have cancelled posting order procedure.",0+48, "Command Warning")
   else
      select sodtl
      set order to order
      
      select sohdr
      go top
      scan for sohdr.lposted
         wait "Please wait, posting " + sohdr.order + " ..." window nowait
         select sodtl
         seek sohdr.order
         scan while sodtl.order = sohdr.order
            select sodtl
            *if empty(sodtl.invoice) and not empty(sodtl.amount)
            *   replace amount with 0
            *endif
            *if empty(sodtl.pv + sodtl.vinvoice) and not empty(sodtl.cost)
            *   replace cost with 0
            *endif
            if empty(amount) and empty(cost) and empty(invoice + pv + vinvoice) 
               delete
            endif
         endscan
         select sohdr
         replace sohdr.posted with thisform.txtWritten.value, sohdr.postedby with left(_userid,10)
      endscan
      =MESSAGEBOX("Posting order procedure is succesfull.",0+48, "Command Warning")
   endif
endif
select sohdr
thisform.release
ENDPROC

PROCEDURE Command1.Click
select sohdr
do Case
   Case _fflag = 'AE'
      do form faeordentry.scx with sohdr.order
   Case _fflag = 'AI'
      do form faiordentry.scx with sohdr.order
   Case _fflag = 'SE'
      do form fseordentry.scx with sohdr.order
   Case _fflag = 'SI'
      do form fsiordentry.scx with sohdr.order
   *Case mdivs = 'DE'
   *   do form fdeordentry.scx with sohdr.order
   *Case mdivs = 'DI'
   *   do form fdiordentry.scx with sohdr.order
   Otherwise
     =MESSAGEBOX("No zoom order for this order.",0+48, "Command Warning")
endcase
select sohdr
thisform.refresh
ENDPROC
